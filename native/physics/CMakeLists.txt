cmake_minimum_required(VERSION 3.22)
project(gg_physics LANGUAGES C CXX)
set(BOX3D_DOUBLE_PRECISION ON CACHE BOOL "Double precision world positions" FORCE)
set(BOX3D_PROFILE OFF CACHE BOOL "Offline build" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_options(-ffp-contract=off)
add_subdirectory(third_party/box3d)
add_library(gg_physics SHARED gg_physics.cpp)
target_compile_features(gg_physics PRIVATE cxx_std_17)
target_compile_options(gg_physics PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden)
target_compile_definitions(gg_physics PRIVATE BOX3D_DOUBLE_PRECISION)
target_link_libraries(gg_physics PRIVATE box3d)
